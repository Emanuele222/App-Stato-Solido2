% Bande della catena biatomica: (a) E_A < E_B, (b) E_A = E_B (§2)
% unita' arbitrarie: (a) E_A=0, E_B=2, t=0.6 ; (b) E_A=E_B=1, t=0.6 ; ascissa x = ka/pi
\begin{tikzpicture}
\pgfplotsset{bia/.style={width=6.6cm, height=6.4cm, xmin=-1, xmax=1, ymin=-1.35, ymax=3.75,
  xtick={-1,0,1}, xticklabels={$-\frac\pi a$,$0$,$\frac\pi a$}, ytick=\empty,
  axis lines=left, clip=false, samples=121, domain=-1:1, enlargelimits=false,
  xlabel={$k$}, every axis x label/.style={at={(axis description cs:1,0)}, anchor=west, xshift=3pt},
  ylabel={$E$}, every axis y label/.style={at={(ticklabel* cs:1.0)}, anchor=south, rotate=0}}}
% ---------------------------------------------------------------- (a)
\begin{axis}[bia, name=pa]
 \fill[bandempty!80] (axis cs:-1,0) rectangle (axis cs:1,2);
 \draw[densely dashed, black!45] (axis cs:-1,0) -- (axis cs:1,0);
 \draw[densely dashed, black!45] (axis cs:-1,2) -- (axis cs:1,2);
 \node[font=\scriptsize, anchor=east] at (axis cs:-1,0) {$E_A$};
 \node[font=\scriptsize, anchor=east] at (axis cs:-1,2) {$E_B$};
 \addplot[very thick, vA] {1+sqrt(1+1.44*cos(pi*x/2)^2)};
 \addplot[very thick, vB] {1-sqrt(1+1.44*cos(pi*x/2)^2)};
 \node[font=\scriptsize, vA, anchor=south west] at (axis cs:0.08,2.6) {conduzione $E_+$};
 \node[font=\scriptsize, vB, anchor=north] at (axis cs:0,-0.62) {valenza $E_-$};
 % gap al bordo e separazione in k=0
 \draw[<->, black!75] (axis cs:0.86,0) -- (axis cs:0.86,2) node[midway, fill=bandempty!80, inner sep=1pt, font=\scriptsize] {$\Eg$};
 \draw[<->, black!55] (axis cs:-0.05,-0.562) -- (axis cs:-0.05,2.562);
 \node[font=\scriptsize, black!70, fill=bandempty!80, inner sep=1pt] at (axis cs:-0.3,1.0) {$\Eg+2W$};
 \draw[<->, black!60] (axis cs:-0.9,2) -- (axis cs:-0.9,2.562);
 \node[font=\scriptsize, black!70, anchor=west] at (axis cs:-0.88,2.28) {$W$};
 \node[font=\small, anchor=north west] at (axis cs:-1,3.75) {(a) $E_A<E_B$};
\end{axis}
% ---------------------------------------------------------------- (b)
\begin{axis}[bia, at={(pa.east)}, anchor=west, xshift=1.6cm]
 \draw[densely dashed, black!45] (axis cs:-1,1) -- (axis cs:1,1);
 \node[font=\scriptsize, anchor=east] at (axis cs:-1,1) {$E_A=E_B$};
 \addplot[very thick, vA] {1+1.2*abs(cos(pi*x/2))};
 \addplot[very thick, vB] {1-1.2*abs(cos(pi*x/2))};
 \fill (axis cs:1,1) circle (1.6pt); \fill (axis cs:-1,1) circle (1.6pt);
 \node[font=\scriptsize, anchor=west, align=left] at (axis cs:0.25,1.45) {nessun gap};
 \draw[->, black!50] (axis cs:0.55,1.3) -- (axis cs:0.95,1.05);
 \node[font=\small, anchor=north west] at (axis cs:-1,3.75) {(b) $E_A=E_B$};
\end{axis}
\end{tikzpicture}
