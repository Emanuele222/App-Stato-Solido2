% ---- orbitali: chiaro = segno +, scuro = segno -
\tikzset{lobop/.style={fill=black!10, draw=black!55, line width=0.4pt},
         lobom/.style={fill=black!60, draw=black!75, line width=0.4pt}}
\def\orbs#1#2{% centro, segno
  \ifnum#2>0 \draw[lobop] (#1) circle (0.25); \node[font=\tiny] at (#1) {$+$};
  \else \draw[lobom] (#1) circle (0.25); \node[font=\tiny, white] at (#1) {$-$}; \fi}
\def\orbp#1#2{% centro, orientazione (+1: lobo + a destra)
  \ifnum#2>0
    \draw[lobom] ($(#1)+(-0.145,0)$) ellipse (0.14 and 0.19); \node[font=\tiny, white] at ($(#1)+(-0.145,0)$) {$-$};
    \draw[lobop] ($(#1)+(0.145,0)$) ellipse (0.14 and 0.19); \node[font=\tiny] at ($(#1)+(0.145,0)$) {$+$};
  \else
    \draw[lobop] ($(#1)+(-0.145,0)$) ellipse (0.14 and 0.19); \node[font=\tiny] at ($(#1)+(-0.145,0)$) {$+$};
    \draw[lobom] ($(#1)+(0.145,0)$) ellipse (0.14 and 0.19); \node[font=\tiny, white] at ($(#1)+(0.145,0)$) {$-$};
  \fi}
